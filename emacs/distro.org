* Introduction

This is a literate Emacs configuration aimed at people coming from Neovim.
It provides:

- Evil + leader key (SPC) with which-key
- Vertico / Consult / Embark / Company (Telescope-like experience)
- Easy language tooling via small =lisp/lang-*.el= files
- Automatic Tree-sitter grammar installation
- Format-on-save with Apheleia
- DAP debugging support
- Magit, Treemacs, vterm, Doom Modeline, Catppuccin, etc.
- Full .NET / Critter Stack support with Docker Compose
  (automatically uses both JetBrains Rider and Visual Studio override files)
- Go, Typescript/Javascript, and Svelte support 
- Swappable package manager (package.el / straight.el / elpaca)
- Doom-inspired startup & runtime performance optimisations

Tangling this file produces:

- =init.el=
- =lisp/lang-python.el=

* Package manager selection

Change the value of =my/package-manager= to switch backends.
Everything else in this config stays the same.

Supported values:
- =package=   → classic package.el (current default)
- =straight=  → straight.el (faster, reproducible)
- =elpaca=    → elpaca (modern async alternative)

* Early init (Doom-inspired performance)

This file is loaded *before* =init.el=. Most of the big startup wins live here.

#+begin_src emacs-lisp :tangle early-init.el
;;; early-init.el --- Early optimisations borrowed from Doom -*- lexical-binding: t; -*-

;; 1. Garbage collection – the single biggest startup win
(setq gc-cons-threshold most-positive-fixnum
      gc-cons-percentage 0.6)

;; 2. Prevent expensive frame resizing when fonts change
(setq frame-inhibit-implied-resize t)

;; 3. Disable UI elements as early as possible
(push '(menu-bar-lines . 0) default-frame-alist)
(push '(tool-bar-lines . 0) default-frame-alist)
(push '(vertical-scroll-bars) default-frame-alist)

;; 4. Silence the startup screen and message
(setq inhibit-startup-screen t
      inhibit-startup-echo-area-message user-login-name
      inhibit-startup-buffer-menu t)

;; 5. Native compilation settings (Emacs 28+)
(when (and (fboundp 'native-comp-available-p)
           (native-comp-available-p))
  (setq native-comp-async-report-warnings-errors 'silent
        native-comp-async-query-on-exit nil
        native-comp-jit-compilation t))

;; 6. Slightly faster file-name handling during startup
(setq file-name-handler-alist nil)

  ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
  ;;; 2. Package management (swappable)
  ;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

  (pcase my/package-manager

    ;; ------------------------------------------------------------------
    ;; Classic package.el
    ;; ------------------------------------------------------------------
    ('package
     (require 'package)
     (setq package-archives
           '(("melpa"  . "https://melpa.org/packages/")
  	   ("melpa-stable" . "https://stable.melpa.org/packages/")
             ("elpa"   . "https://elpa.gnu.org/packages/")
             ("nongnu" . "https://elpa.nongnu.org/nongnu/")))
     (package-initialize)
     (unless package-archive-contents
       (package-refresh-contents))

     (unless (package-installed-p 'use-package)
       (package-install 'use-package))
     (require 'use-package)
     (setq use-package-always-ensure t))

    ;; ------------------------------------------------------------------
    ;; straight.el (recommended faster alternative)
    ;; ------------------------------------------------------------------
    ('straight
     (defvar bootstrap-version)
     (let ((bootstrap-file
            (expand-file-name
             "straight/repos/straight.el/bootstrap.el"
             (or (bound-and-true-p straight-base-dir)
                 user-emacs-directory)))
           (bootstrap-version 7))
       (unless (file-exists-p bootstrap-file)
         (with-current-buffer
             (url-retrieve-synchronously
              "https://raw.githubusercontent.com/radian-software/straight.el/develop/install.el"
              'silent 'inhibit-cookies)
           (goto-char (point-max))
           (eval-print-last-sexp)))
       (load bootstrap-file nil 'nomessage))

     (straight-use-package 'use-package)
     (setq straight-use-package-by-default t
           use-package-always-ensure nil))

    ;; ------------------------------------------------------------------
    ;; elpaca (modern async alternative)
    ;; ------------------------------------------------------------------
    ('elpaca
     (defvar elpaca-installer-version 0.11)
     (defvar elpaca-directory (expand-file-name "elpaca/" user-emacs-directory))
     (defvar elpaca-builds-directory (expand-file-name "builds/" elpaca-directory))
     (defvar elpaca-repos-directory (expand-file-name "repos/" elpaca-directory))
     (defvar elpaca-order
       '(elpaca :repo "https://github.com/progfolio/elpaca.git"
                :ref nil :depth 1 :inherit nil
                :files (:defaults "elpaca-test.el" (:exclude "extensions"))
                :build (:not elpaca--activate-package)))
     (let* ((repo  (expand-file-name "elpaca/" elpaca-repos-directory))
            (build (expand-file-name "elpaca/" elpaca-builds-directory))
            (order (cdr elpaca-order))
            (default-directory repo))
       (add-to-list 'load-path (if (file-exists-p build) build repo))
       (unless (file-exists-p repo)
         (make-directory repo t)
         (when (<= emacs-major-version 28) (require 'subr-x))
         (condition-case-unless-debug err
             (if-let* ((buffer (pop-to-buffer-same-window "*elpaca-bootstrap*"))
                       ((zerop (apply #'call-process `("git" nil ,buffer t "clone"
                                                      ,@(when-let* ((depth (plist-get order :depth)))
                                                          (list (format "--depth=%d" depth) "--no-single-branch"))
                                                      ,(plist-get order :repo) ,repo))))
                       ((zerop (call-process "git" nil buffer t "checkout"
                                             (or (plist-get order :ref) "--"))))
                       (emacs (concat invocation-directory invocation-name))
                       ((zerop (call-process emacs nil buffer nil "-Q" "-L" "." "--batch"
                                             "--eval" "(byte-recompile-directory \".\" 0 'force)")))
                       ((require 'elpaca))
                       ((elpaca-generate-autoloads "elpaca" repo)))
                 (progn (message "%s" (buffer-string)) (kill-buffer buffer))
               (error "%s" (with-current-buffer buffer (buffer-string))))
           ((error) (delete-directory repo 'recursive)
                    (error "%s" (error-message-string err)))))
       (unless (require 'elpaca-autoloads nil t)
         (require 'elpaca)
         (elpaca-generate-autoloads "elpaca" repo)
         (load "./elpaca-autoloads")))
     (add-hook 'after-init-hook #'elpaca-process-queues)
     (elpaca `(,@elpaca-order))

     (elpaca elpaca-use-package
       (elpaca-use-package-mode)
       (setq use-package-always-ensure t)))

    (_ (error "Unknown package manager: %s" my/package-manager)))

* Startup & runtime performance (Doom-inspired)

#+begin_src emacs-lisp :tangle init.el
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;;; Performance – borrowed from Doom Emacs
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;; Remember the original file-name-handler-alist so we can restore it
(defvar my--file-name-handler-alist file-name-handler-alist)

;; Restore a sane GC threshold after startup
(add-hook 'emacs-startup-hook
          (lambda ()
            (setq gc-cons-threshold (* 32 1024 1024)  ; 32 MB
                  gc-cons-percentage 0.1
                  file-name-handler-alist
                  (append (bound-and-true-p my--file-name-handler-alist)
                          file-name-handler-alist))))

;; gcmh – “Garbage Collector Magic Hack” (Doom uses this)
(use-package gcmh
  :demand t
  :config
  (setq gcmh-idle-delay 'auto
        gcmh-auto-idle-delay-factor 10
        gcmh-high-cons-threshold (* 64 1024 1024)) ; 64 MB
  (gcmh-mode 1))

;; Slightly nicer behaviour while the minibuffer is active
(defun my/defer-gc-during-minibuffer ()
  (setq gc-cons-threshold most-positive-fixnum))
(defun my/restore-gc-after-minibuffer ()
  (run-at-time 1 nil
               (lambda () (setq gc-cons-threshold (* 32 1024 1024)))))
(add-hook 'minibuffer-setup-hook #'my/defer-gc-during-minibuffer)
(add-hook 'minibuffer-exit-hook  #'my/restore-gc-after-minibuffer)

;; Prefer fundamental-mode for the initial scratch buffer
(setq initial-major-mode 'fundamental-mode)

(setq inhibit-compacting-font-caches t)

(setq redisplay-skip-fontification-on-input t)

* How to use this file

1. Save this buffer as =\~/.emacs.d/config.org=
2. Run =M-x org-babel-tangle= (or =C-c C-v t=)
3. This creates/updates:
   - =early-init.el=
   - =init.el=
   - =lisp/lang-*.el=
4. Restart Emacs

**Switching package manager later**

Just change one line near the top:

#+begin_src emacs-lisp
(defvar my/package-manager 'straight)   ; ← change to 'straight or 'elpaca
#+end_src

Then tangle + restart Emacs.
